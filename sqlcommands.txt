-- Task 1: Create database school_db
CREATE DATABASE school_db;

-- Task 2: Create table students in school_db
USE school_db;

CREATE TABLE students (
    id INT,
    name VARCHAR(50),
    age INT
);

-- Task 3: Create database company_db
CREATE DATABASE company_db;

-- Task 4 & Task 5: Create tables in company_db
USE company_db;

CREATE TABLE employees (
    emp_id INT,
    emp_name VARCHAR(100),
    salary DECIMAL(10,2),
    department VARCHAR(50)
);

CREATE TABLE products (
    product_id INT,
    product_name VARCHAR(100),
    price FLOAT
);

-- Task 6: Create database library_db
CREATE DATABASE library_db;

-- Task 7 & Task 8: Create tables in library_db
USE library_db;

CREATE TABLE books (
    book_id INT,
    title VARCHAR(150),
    author VARCHAR(100),
    price DECIMAL(8,2)
);

CREATE TABLE orders (
    order_id INT,
    customer_name VARCHAR(100),
    order_date DATE,
    amount DECIMAL(10,2)
);
